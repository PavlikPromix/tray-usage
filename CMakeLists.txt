cmake_minimum_required(VERSION 3.20)
project(TrayUsage VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "TrayUsage requires Windows 10 or newer")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(TrayUsage WIN32 src/main.cpp src/app.rc)
target_include_directories(TrayUsage PRIVATE src)
target_compile_definitions(TrayUsage PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(TrayUsage PRIVATE winhttp shell32 user32 gdi32 advapi32)
if(MSVC)
  target_compile_options(TrayUsage PRIVATE /W4 /WX /permissive- /utf-8 /GS /guard:cf $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL>)
  target_link_options(TrayUsage PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf $<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF,ICF>)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(usage_tests tests/usage_tests.cpp)
  if(MSVC)
    target_compile_options(usage_tests PRIVATE /W4 /WX /permissive- /utf-8)
  endif()
  add_test(NAME usage_tests COMMAND usage_tests)
  add_executable(native_tests tests/native_tests.cpp)
  target_compile_definitions(native_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(native_tests PRIVATE winhttp shell32 user32 gdi32 advapi32)
  if(MSVC)
    target_compile_options(native_tests PRIVATE /W4 /WX /permissive- /utf-8)
  endif()
  add_test(NAME native_tests COMMAND native_tests)
endif()
